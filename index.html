        function exportToCSV(type) {
            var tables = [];
            var filename = '';
            if (type === 'new') {
                tables = [document.getElementById('newToolTable')];
                filename = 'New_CARE_Tool_Implementation_Activities.csv';
            } else if (type === 'ongoing') {
                tables = [document.getElementById('ongoingToolTable')];
                filename = 'Ongoing_CARE_Tool_Maintenance_Activities.csv';
            } else {
                tables = [document.getElementById('newToolTable'), document.getElementById('ongoingToolTable')];
                filename = 'Complete_CARE_Tools_CQI_Timeline.csv';
            }
            var csv = [];
            tables.forEach(function(table, index) {
                if (index > 0) { csv.push(''); csv.push(''); }
                if (type === 'all') {
                    csv.push(index === 0 ? 'NEW CARE TOOL DEVELOPMENT & INITIAL IMPLEMENTATION' : 'ONGOING CARE TOOL MAINTENANCE & CQI');
                    csv.push('');
                }
                var rows = table.querySelectorAll('tr');
                var headers = Array.from(rows[0].querySelectorAll('th')).map(function(th) { return th.textContent.trim(); });
                csv.push(headers.join(','));
                for (var i = 1; i < rows.length; i++) {
                    var cols = Array.from(rows[i].querySelectorAll('td')).map(function(td) {
                        var text = td.textContent.trim().replace(/\s+/g, ' ').replace(/"/g, '""');
                        return '"' + text + '"';
                    });
                    csv.push(cols.join(','));
                }
            });
            var csvContent = csv.join('\n');
            var blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            var link = document.createElement('a');
            var url = URL.createObjectURL(blob);
            link.setAttribute('href', url);
            link.setAttribute('download', filename);
            link.style.visibility = 'hidden';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
    </script>

        function exportToCSV(type) {
            var tables = [];
            var filename = '';
            if (type === 'new') {
                tables = [document.getElementById('newToolTable')];
                filename = 'New_CARE_Tool_Implementation_Activities.csv';
            } else if (type === 'ongoing') {
                tables = [document.getElementById('ongoingToolTable')];
                filename = 'Ongoing_CARE_Tool_Maintenance_Activities.csv';
            } else {
                tables = [document.getElementById('newToolTable'), document.getElementById('ongoingToolTable')];
                filename = 'Complete_CARE_Tools_CQI_Timeline.csv';
            }
            var csv = [];
            tables.forEach(function(table, index) {
                if (index > 0) { csv.push(''); csv.push(''); }
                if (type === 'all') {
                    csv.push(index === 0 ? 'NEW CARE TOOL DEVELOPMENT & INITIAL IMPLEMENTATION' : 'ONGOING CARE TOOL MAINTENANCE & CQI');
                    csv.push('');
                }
                var rows = table.querySelectorAll('tr');
                var headers = Array.from(rows[0].querySelectorAll('th')).map(function(th) { return th.textContent.trim(); });
                csv.push(headers.join(','));
                for (var i = 1; i < rows.length; i++) {
                    var cols = Array.from(rows[i].querySelectorAll('td')).map(function(td) {
                        var text = td.textContent.trim().replace(/\s+/g, ' ').replace(/"/g, '""');
                        return '"' + text + '"';
                    });
                    csv.push(cols.join(','));
                }
            });
            var csvContent = csv.join('\n');
            var blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            var link = document.createElement('a');
            var url = URL.createObjectURL(blob);
            link.setAttribute('href', url);
            link.setAttribute('download', filename);
            link.style.visibility = 'hidden';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
    </script>
